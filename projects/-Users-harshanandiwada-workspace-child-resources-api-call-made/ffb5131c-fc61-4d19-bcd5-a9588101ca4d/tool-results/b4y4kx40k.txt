                "id": channel_id,

                "warehouse_customer_id": wh_customer_id,
                "additional_fields": {},

                assert existing_channel == formatted_order_channel, (

        if not first_loop:
            return {"data": []}
        # ShipmentOrder responses only carry channelId + client ID,
        # but many channel setups can share the same channelId
        # (differing only by setupName)
        # Pull the full channel-setup list and dedup by
        # channelId + client ID
        # when we were doing this just from orders, name wasn't always filled out
        channels_by_id = {}
        page_token = 0
        has_next_page = True
        while has_next_page:
            path = f"DataSetup/channelsetup/list/i/{page_token}/s/{PAGE_SIZE}"
            rsp = self.logiwa_request(path)
            # observed: KIBO connection returns {} — no channelsetup access
            if not rsp or "totalCount" not in rsp:
                break
            has_next_page = False
            if rsp["totalCount"] > (page_token + 1) * PAGE_SIZE:
                page_token += 1
                has_next_page = True
            for ch in rsp["data"]:
                channel_id = str(ch["channelId"]) if ch["channelId"] else None
                if not channel_id:
                    continue
                client_id = ch["clientIdentifier"]
                channel_name = ch["channelName"]
                full_channel_id = f"{channel_id}_{client_id}"

                    "id": full_channel_id,
                    "name": channel_name,

                    "additional_fields": {},
